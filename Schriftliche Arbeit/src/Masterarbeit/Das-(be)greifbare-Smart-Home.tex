% Monochromes Code-Listing nach dem Listing-Aufbau des Handbuchs.
\newcommand{\CodeFile}[3][]{%
    \refstepcounter{lstlisting}%
    \ifblank{#1}{%
        \protected@edef\@currentlabelname{Listing \thelstlisting}%
    }{%
        \addcontentsline{lol}{lstlisting}{\protect\numberline{\thelstlisting}#1}%
        \protected@edef\@currentlabelname{#1}%
    }%
    \tcbinputlisting{%
        listing file={#3},
        listing options={
            language=#2,
            basicstyle=\ttfamily\small\color{black},
            keywordstyle=\bfseries\color{black},
            stringstyle=\color{black},
            commentstyle=\itshape\color{black},
            numbers=left,
            numberstyle=\tiny\sffamily\color{black}\nocopy,
            numbersep=12pt,
            showstringspaces=false,
            breaklines=true,
            tabsize=2
        },
        listing only,
        enhanced,
        width=\linewidth-1.6cm,
        enlarge left by=0.8cm,
        arc=0pt,
        outer arc=0pt,
        colback=white,
        colframe=black,
        boxrule=0.5pt,
        bottomrule=0.5pt,
        left=2mm, right=4mm, top=2mm, bottom=2mm,
        fonttitle=\sffamily\bfseries\footnotesize,
        title={Listing~\thelstlisting\ifblank{#1}{}{: #1} \hfill \texttt{#2}},
        after={\vspace{1.2em}},
    }%
    \ifblank{#1}{%
        \protected@edef\@currentlabelname{Listing \thelstlisting}%
    }{%
        \protected@edef\@currentlabelname{#1}%
    }%
}
\makeatother

% Bildbefehl nach dem Handbuch-Aufbau, vollständig in Schwarzweiß.
\newcommand{\KeyImage}[4][]{%
    \begin{figure}[H]
        \centering
        \sbox0{%
            \ifblank{#1}{%
                \includegraphics[width=\dimexpr 0.86\textwidth\relax]{#4}%
            }{%
                \includegraphics[#1]{#4}%
            }%
        }%
        \begin{tcolorbox}[
            enhanced,
            width=\dimexpr\wd0 + 8mm\relax,
            colframe=black,
            colback=white,
            arc=0pt,
            boxrule=0.5pt,
            left=2mm, right=2mm, top=2mm, bottom=1.5mm,
            nobeforeafter
        ]
            \centering
            \usebox{0}%
            \ifblank{#2}{}{%
                \vspace{0.3em}
                {\tiny\sffamily\color{black}Quelle: #2\par}%
            }%
        \end{tcolorbox}
        \caption{#3}
    \end{figure}%
}
